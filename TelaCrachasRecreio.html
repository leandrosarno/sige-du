function gerarCrachasRecreio(dados) { let ssCeu, abaImpCriancas, abaImpPais, tempMoldeCrianca, tempMoldePais, abaRegrasCeu; try { ssCeu = SpreadsheetApp.openById(dados.idPlanilha); const ssMaster = SpreadsheetApp.getActiveSpreadsheet(); const abaCriancas = ssCeu.getSheetByName('RF - CRIANCAS') || ssCeu.getSheetByName('RF - CRIANÇAS'); const abaRegrasMaster = ssMaster.getSheetByName('RF - REGRAS') || ssMaster.getSheetByName('RF - REGRA'); if (!abaCriancas || !abaRegrasMaster) return { sucesso: false, erro: "Abas necessárias não encontradas." }; const idOuro = String(abaRegrasMaster.getRange("B42").getValue()).trim(); let ssOuro; try { ssOuro = SpreadsheetApp.openById(idOuro); } catch(e) { return { sucesso: false, erro: "Falha ao abrir a Planilha Ouro." }; } let moldeCrianca = ssOuro.getSheetByName('Crachá das Crianças') || ssOuro.getSheetByName('RF - CRACHA'); let moldePais = ssOuro.getSheetByName('Crachá dos Pais') || ssOuro.getSheetByName('RF - CRACHA PAIS'); // Limpeza rigorosa inicial ssCeu.getSheets().forEach(s => { let n = s.getName(); if (n === "Impressão Crachás" || n === "Impressão Pais" || n.startsWith("TEMP_")) ssCeu.deleteSheet(s); }); tempMoldeCrianca = moldeCrianca.copyTo(ssCeu); tempMoldeCrianca.setName("TEMP_MOLDE_CRIANCAS"); tempMoldePais = moldePais.copyTo(ssCeu); tempMoldePais.setName("TEMP_MOLDE_PAIS"); abaRegrasCeu = abaRegrasMaster.copyTo(ssCeu); abaRegrasCeu.setName("TEMP_REGRAS"); const regrasIdade = { 4: abaRegrasCeu.getRange("B29").getBackground(), 5: abaRegrasCeu.getRange("B30").getBackground(), 6: abaRegrasCeu.getRange("B31").getBackground(), 7: abaRegrasCeu.getRange("B32").getBackground(), 8: abaRegrasCeu.getRange("B33").getBackground(), 9: abaRegrasCeu.getRange("B34").getBackground(), 10: abaRegrasCeu.getRange("B35").getBackground(), 11: abaRegrasCeu.getRange("B36").getBackground(), 12: abaRegrasCeu.getRange("B37").getBackground(), 13: abaRegrasCeu.getRange("B38").getBackground(), 14: abaRegrasCeu.getRange("B39").getBackground() }; const corVaiSozinhoSim = abaRegrasCeu.getRange("D23").getBackground(); const corVaiSozinhoNao = abaRegrasCeu.getRange("E23").getBackground(); const formatarData = (val) => (val instanceof Date) ? Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy") : String(val); const textoPeriodo = String(abaRegrasCeu.getRange("B12").getValue()).trim() + " - PERÍODO DE " + formatarData(abaRegrasCeu.getRange("B13").getValue()) + " ATÉ " + formatarData(abaRegrasCeu.getRange("B14").getValue()); const dadosCr = abaCriancas.getDataRange().getValues(); const fila = []; for (let i = 1; i < dadosCr.length; i++) { let imprimir = String(dadosCr[i][8]).toUpperCase().trim(); if (imprimir === "SIM" || imprimir === "TRUE" || dadosCr[i][8] === true) { fila.push({ nome: String(dadosCr[i][2]).toUpperCase(), idade: dadosCr[i][3], resp: String(dadosCr[i][5]).toUpperCase(), tel: dadosCr[i][6], restImg: String(dadosCr[i][9]).toUpperCase() === "SIM", dieta: String(dadosCr[i][10]).toUpperCase() === "SIM", sozinho: String(dadosCr[i][11]).toUpperCase() === "SIM", inclusao: String(dadosCr[i][12]).toUpperCase() === "SIM", piscina: String(dadosCr[i][13]).toUpperCase() === "SIM" }); } } if (fila.length === 0) { return { sucesso: false, erro: "Nenhuma criança marcada com SIM." }; } // ==================================================================== // MÓDULO 1: CRACHÁS CRIANÇAS (Restaurado com seu código exato) // ==================================================================== abaImpCriancas = ssCeu.insertSheet("Impressão Crachás"); abaImpCriancas.setHiddenGridlines(true); if (abaImpCriancas.getMaxColumns() < 34) { abaImpCriancas.insertColumnsAfter(abaImpCriancas.getMaxColumns(), 34 - abaImpCriancas.getMaxColumns()); } const passoColunaC = 17; const passoLinhaC = 14; let largurasC = []; for (let c = 1; c <= passoColunaC; c++) { largurasC.push(tempMoldeCrianca.getColumnWidth(c)); } let alturasC = []; for (let r = 1; r <= passoLinhaC; r++) { alturasC.push(tempMoldeCrianca.getRowHeight(r)); } for (let c = 1; c <= 16; c++) { abaImpCriancas.setColumnWidth(c, largurasC[c - 1] || 100); abaImpCriancas.setColumnWidth(c + passoColunaC, largurasC[c - 1] || 100); } abaImpCriancas.setColumnWidth(passoColunaC, largurasC[passoColunaC - 1] || 15); let crachasNaPaginaC = 0; let paginaAtualC = 1; for (let i = 0; i < fila.length; i++) { let c = fila[i]; let colGrid = crachasNaPaginaC % 2; let linGrid = Math.floor(crachasNaPaginaC / 2); let offsetLinha = (paginaAtualC - 1) * (4 * passoLinhaC) + (linGrid * passoLinhaC) + 1; let offsetColuna = (colGrid * passoColunaC) + 1; if (abaImpCriancas.getMaxRows() < offsetLinha + passoLinhaC) { abaImpCriancas.insertRowsAfter(abaImpCriancas.getMaxRows(), 20); } tempMoldeCrianca.getRange("A1:P13").copyTo(abaImpCriancas.getRange(offsetLinha, offsetColuna)); if (colGrid === 0) { for (let r = 1; r <= 13; r++) { abaImpCriancas.setRowHeight(offsetLinha + r - 1, alturasC[r - 1] || 21); } abaImpCriancas.setRowHeight(offsetLinha + 13, 20); } // Restauração das injeções usando os blocos grandes originais abaImpCriancas.getRange(offsetLinha + 1, offsetColuna + 4, 1, 12).setValue(String(dados.nomeCeu).toUpperCase()); abaImpCriancas.getRange(offsetLinha + 2, offsetColuna + 4, 1, 12).setValue(textoPeriodo); abaImpCriancas.getRange(offsetLinha + 3, offsetColuna, 4, 16).setValue(c.nome); if (regrasIdade[c.idade]) { abaImpCriancas.getRange(offsetLinha + 3, offsetColuna, 4, 16).setBackground(regrasIdade[c.idade]); } abaImpCriancas.getRange(offsetLinha + 8, offsetColuna, 1, 8).setValue(c.resp); abaImpCriancas.getRange(offsetLinha + 8, offsetColuna + 8, 1, 4).setValue(c.tel); abaImpCriancas.getRange(offsetLinha + 7, offsetColuna + 12, 6, 4).setValue(c.idade); let corFundoVaiSozinho = c.sozinho ? corVaiSozinhoSim : corVaiSozinhoNao; abaImpCriancas.getRange(offsetLinha + 11, offsetColuna, 2, 12).setBackground(corFundoVaiSozinho); // Cópia física direta dos ícones do seu código antigo let rImg = c.restImg ? "B21" : "C21"; abaRegrasCeu.getRange(rImg).copyTo(abaImpCriancas.getRange(offsetLinha + 9, offsetColuna, 2, 3)); let rDieta = c.dieta ? "B22" : "C22"; abaRegrasCeu.getRange(rDieta).copyTo(abaImpCriancas.getRange(offsetLinha + 9, offsetColuna + 3, 2, 3)); let rInc = c.inclusao ? "B24" : "C24"; abaRegrasCeu.getRange(rInc).copyTo(abaImpCriancas.getRange(offsetLinha + 9, offsetColuna + 6, 2, 3)); let rPisc = c.piscina ? "B25" : "C25"; abaRegrasCeu.getRange(rPisc).copyTo(abaImpCriancas.getRange(offsetLinha + 9, offsetColuna + 9, 2, 3)); crachasNaPaginaC++; if (crachasNaPaginaC === 8) { crachasNaPaginaC = 0; paginaAtualC++; } } // ==================================================================== // MÓDULO 2: CRACHÁS DOS PAIS // ==================================================================== abaImpPais = ssCeu.insertSheet("Impressão Pais"); abaImpPais.setHiddenGridlines(true); if (abaImpPais.getMaxColumns() < 34) { abaImpPais.insertColumnsAfter(abaImpPais.getMaxColumns(), 34 - abaImpPais.getMaxColumns()); } const passoColunaP = 17; const passoLinhaP = 10; let largurasP = []; for (let c = 1; c <= passoColunaP; c++) { largurasP.push(tempMoldePais.getColumnWidth(c)); } let alturasP = []; for (let r = 1; r <= passoLinhaP; r++) { alturasP.push(tempMoldePais.getRowHeight(r)); } for (let c = 1; c <= 16; c++) { abaImpPais.setColumnWidth(c, largurasP[c - 1] || 100); abaImpPais.setColumnWidth(c + passoColunaP, largurasP[c - 1] || 100); } abaImpPais.setColumnWidth(passoColunaP, largurasP[passoColunaP - 1] || 15); let crachasNaPaginaP = 0; let paginaAtualP = 1; for (let i = 0; i < fila.length; i++) { let c = fila[i]; let colGrid = crachasNaPaginaP % 2; let linGrid = Math.floor(crachasNaPaginaP / 2); let offsetLinha = (paginaAtualP - 1) * (5 * passoLinhaP) + (linGrid * passoLinhaP) + 1; let offsetColuna = (colGrid * passoColunaP) + 1; if (abaImpPais.getMaxRows() < offsetLinha + passoLinhaP) { abaImpPais.insertRowsAfter(abaImpPais.getMaxRows(), 20); } tempMoldePais.getRange("A1:P9").copyTo(abaImpPais.getRange(offsetLinha, offsetColuna)); if (colGrid === 0) { for (let r = 1; r <= 9; r++) { abaImpPais.setRowHeight(offsetLinha + r - 1, alturasP[r - 1] || 21); } abaImpPais.setRowHeight(offsetLinha + 9, 20); } abaImpPais.getRange(offsetLinha + 2, offsetColuna + 3, 1, 13).setValue(textoPeriodo); abaImpPais.getRange(offsetLinha + 3, offsetColuna, 4, 13).setValue(c.nome); abaImpPais.getRange(offsetLinha + 8, offsetColuna, 1, 16).setValue(c.resp); abaImpPais.getRange(offsetLinha + 3, offsetColuna + 13, 4, 3).setValue(c.idade); crachasNaPaginaP++; if (crachasNaPaginaP === 10) { crachasNaPaginaP = 0; paginaAtualP++; } } SpreadsheetApp.flush(); Utilities.sleep(2000); // ==================================================================== // EXPORTAÇÃO NATIVA (ZIP) // ==================================================================== const token = ScriptApp.getOAuthToken(); const urlBase = "https://docs.google.com/spreadsheets/d/" + ssCeu.getId() + "/export?"; const opcoesComuns = "exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&gridlines=false&printtitle=false&sheetnames=false&pagenumbers=false&attachment=false&top_margin=0.20&bottom_margin=0.20&left_margin=0.20&right_margin=0.20&gid="; let resC = UrlFetchApp.fetch(urlBase + opcoesComuns + abaImpCriancas.getSheetId(), { headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true }); let blobC = resC.getBlob().setName("1_CRACHAS_CRIANCAS.pdf"); let resP = UrlFetchApp.fetch(urlBase + opcoesComuns + abaImpPais.getSheetId(), { headers: { 'Authorization': 'Bearer ' + token }, muteHttpExceptions: true }); let blobP = resP.getBlob().setName("2_CRACHAS_PAIS.pdf"); let zipFile = Utilities.zip([blobC, blobP], "CRACHAS_RECREIO_" + dados.nomeCeu + ".zip"); let arquivo = DriveApp.getRootFolder().createFile(zipFile); arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return { sucesso: true, quantidade: fila.length, paginas: [], zip: arquivo.getDownloadUrl() }; } catch (e) { return { sucesso: false, erro: e.message }; } finally { // A GARANTIA DE LIMPEZA: Este bloco sempre roda, mesmo se der erro no meio da função if (ssCeu) { try { if (abaImpCriancas) ssCeu.deleteSheet(abaImpCriancas); } catch(e){} try { if (abaImpPais) ssCeu.deleteSheet(abaImpPais); } catch(e){} try { if (tempMoldeCrianca) ssCeu.deleteSheet(tempMoldeCrianca); } catch(e){} try { if (tempMoldePais) ssCeu.deleteSheet(tempMoldePais); } catch(e){} try { if (abaRegrasCeu) ssCeu.deleteSheet(abaRegrasCeu); } catch(e){} } } }
